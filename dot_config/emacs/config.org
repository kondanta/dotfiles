  (use-package apheleia

    ;; --- 1. Async LSP Bridge ---
    (cl-defun my/apheleia-lsp-async (&key buffer scratch callback &allow-other-keys)
      "Format BUFFER using LSP asynchronously.
       When LSP returns edits, apply them to SCRATCH and call CALLBACK."
      (with-current-buffer buffer
        (if (and (bound-and-true-p lsp-mode)
                 (lsp-feature? "textDocument/formatting"))
            ;; Send the request asynchronously
            (lsp-request-async
             "textDocument/formatting"
             (lsp--make-document-formatting-params)
             ;; Success Handler: runs when LSP replies
             (lambda (edits)
               (when edits
                 (with-current-buffer scratch
                   (lsp--apply-text-edits edits scratch)))
               ;; CRITICAL: Tell Apheleia we are done
               (funcall callback)))
          ;; If LSP isn't active, just finish immediately
          (funcall callback))))
